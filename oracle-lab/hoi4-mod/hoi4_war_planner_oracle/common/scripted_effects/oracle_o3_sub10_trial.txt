# HOI4 War Planner Oracle O3 sub-10 defended-incidence probe.
# Separate from O1/O2: distinct WPO3 prefix, namespace, scenario, and flags.
#
# Workflow:
# 1. load the exact clean O1 baseline save while paused at 11:00;
# 2. run d_oracle_o3_prepare;
# 3. run built-in random_seed;
# 4. issue the same GER attack while paused;
# 5. run d_oracle_o3_trial6 while still paused; this records h0;
# 6. unpause only until the battle instantiates, then pause immediately during
#    the Oracle-validated h0->h1 no-damage startup interval;
# 7. capture the battle panel and accept only if displayed GER Soft Attack is
#    7 or 8 and remains at least 10 points below POL Defense;
# 8. resume through 17:00. Cleanup is automatic at hour 6.

oracle_o3_log_attacker_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO3 ATTACKER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o3_log_defender_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO3 DEFENDER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o3_capture_sample = {
    log = "WPO3 SAMPLE hour=[?oracle_o3_hour]"
    oracle_o3_log_attacker_divisions = yes
    POL = {
        oracle_o3_log_defender_divisions = yes
    }
}

d_oracle_o3_prepare = {
    GER = {
        clr_country_flag = oracle_o3_prepared
        clr_country_flag = oracle_o3_running

        if = {
            limit = {
                has_dynamic_modifier = {
                    modifier = oracle_o3_sub10_attack_attenuator
                }
            }
            remove_dynamic_modifier = {
                modifier = oracle_o3_sub10_attack_attenuator
            }
            force_update_dynamic_modifier = yes
        }

        add_dynamic_modifier = {
            modifier = oracle_o3_sub10_attack_attenuator
        }
        force_update_dynamic_modifier = yes

        if = {
            limit = {
                has_dynamic_modifier = {
                    modifier = oracle_o3_sub10_attack_attenuator
                }
            }
            set_country_flag = oracle_o3_prepared
            log = "WPO3 STATUS state=prepared scenario=o3-sub10-defended-incidence-v1 attenuator=army_infantry_attack_factor:-0.89 attenuatorPresent=yes expectedRegime=GER-soft-attack-display-7-or-8-sub10"
        }
        else = {
            clr_country_flag = oracle_o3_prepared
            log = "WPO3 STATUS state=prepare-failed scenario=o3-sub10-defended-incidence-v1 attenuatorPresent=no"
        }
    }
}

d_oracle_o3_clear = {
    GER = {
        if = {
            limit = {
                has_dynamic_modifier = {
                    modifier = oracle_o3_sub10_attack_attenuator
                }
            }
            remove_dynamic_modifier = {
                modifier = oracle_o3_sub10_attack_attenuator
            }
            force_update_dynamic_modifier = yes
        }

        clr_country_flag = oracle_o3_prepared
        clr_country_flag = oracle_o3_running

        if = {
            limit = {
                has_dynamic_modifier = {
                    modifier = oracle_o3_sub10_attack_attenuator
                }
            }
            log = "WPO3 STATUS state=clear-failed attenuatorRemoved=no"
        }
        else = {
            log = "WPO3 STATUS state=cleared attenuatorRemoved=yes"
        }
    }
}

oracle_o3_trial6_start = {
    if = {
        limit = {
            has_country_flag = oracle_o3_prepared
            has_dynamic_modifier = {
                modifier = oracle_o3_sub10_attack_attenuator
            }
        }

        clr_country_flag = oracle_o3_running
        set_country_flag = oracle_o3_running
        set_variable = { oracle_o3_hour = 0 }

        log = "WPO3 BEGIN schema=1 scenario=o3-sub10-defended-incidence-v1 gameVersion=1.19.3.0.c01a checksum=5632 checksumScope=base-game-reference method=bisection14 runMode=trial6 tacticMode=neutral-basic-only attenuator=army_infantry_attack_factor:-0.89 prepared=yes attenuatorPresent=yes"
        oracle_o3_capture_sample = yes

        country_event = { id = oracle_o3.1 hours = 1 }
        country_event = { id = oracle_o3.1 hours = 2 }
        country_event = { id = oracle_o3.1 hours = 3 }
        country_event = { id = oracle_o3.1 hours = 4 }
        country_event = { id = oracle_o3.1 hours = 5 }
        country_event = { id = oracle_o3.1 hours = 6 }
    }
    else = {
        log = "WPO3 STATUS state=not-prepared-or-attenuator-missing action=run-d_oracle_o3_prepare-first"
    }
}

d_oracle_o3_trial6 = {
    GER = {
        oracle_o3_trial6_start = yes
    }
}
