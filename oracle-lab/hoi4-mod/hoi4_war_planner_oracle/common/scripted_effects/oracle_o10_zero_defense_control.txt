# O10 zero-defense hit-gate control.
#
# Purpose:
# Keep the exact O8 hit-gate bundle (defended points miss, undefended points hit)
# but force POL Defense to exactly 0. If the O7 1x/2x/3x law reappears, O8's
# mismatch is specifically caused by the defense split rather than by the
# BASE_CHANCE_TO_AVOID_HIT=100 diagnostic setting itself.
#
# Acceptance:
# - GER displayed/tooltip Soft Attack exactly 20.0
# - POL displayed/tooltip Defense exactly 0.0
# - one GER vs one POL 18-width division
# - no commanders, zero reserves

oracle_o10_log_attacker_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO10 ATTACKER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o10_log_defender_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO10 DEFENDER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o10_capture_sample = {
    log = "WPO10 SAMPLE hour=[?oracle_o10_hour]"
    oracle_o10_log_attacker_divisions = yes
    POL = { oracle_o10_log_defender_divisions = yes }
}

d_oracle_o10_prepare = {
    GER = {
        clr_country_flag = oracle_o10_prepared
        clr_country_flag = oracle_o10_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o10_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o10_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o10_attack_target }
        force_update_dynamic_modifier = yes
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o10_defense_zero } }
            remove_dynamic_modifier = { modifier = oracle_o10_defense_zero }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o10_defense_zero }
        force_update_dynamic_modifier = yes
    }
    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o10_attack_target }
                POL = { has_dynamic_modifier = { modifier = oracle_o10_defense_zero } }
            }
            set_country_flag = oracle_o10_prepared
            log = "WPO10 STATUS state=prepared scenario=o10-zero-defense-hit-gate-control-v1 attackModifier=army_infantry_attack_factor:-0.721 defenseModifier=army_infantry_defence_factor:-1.0328836429 modifiersPresent=yes expectedPanel=GER-soft-20-tooltip-20.0-POL-defense-0-tooltip-0.0"
        }
        else = { log = "WPO10 STATUS state=prepare-failed" }
    }
}

d_oracle_o10_clear = {
    GER = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o10_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o10_attack_target }
            force_update_dynamic_modifier = yes
        }
        clr_country_flag = oracle_o10_prepared
        clr_country_flag = oracle_o10_running
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o10_defense_zero } }
            remove_dynamic_modifier = { modifier = oracle_o10_defense_zero }
            force_update_dynamic_modifier = yes
        }
    }
    GER = {
        if = {
            limit = {
                OR = {
                    has_dynamic_modifier = { modifier = oracle_o10_attack_target }
                    POL = { has_dynamic_modifier = { modifier = oracle_o10_defense_zero } }
                }
            }
            log = "WPO10 STATUS state=clear-failed modifiersRemoved=no"
        }
        else = { log = "WPO10 STATUS state=cleared modifiersRemoved=yes" }
    }
}
