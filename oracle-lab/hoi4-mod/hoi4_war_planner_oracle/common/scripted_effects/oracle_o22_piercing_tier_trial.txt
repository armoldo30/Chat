oracle_o22_apply_controls = {
    GER = {
        clr_country_flag = oracle_o22_prepared
        clr_country_flag = oracle_o22_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o22_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o22_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o22_attack_target }
        force_update_dynamic_modifier = yes
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o22_defender_control } }
            remove_dynamic_modifier = { modifier = oracle_o22_defender_control }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o22_defender_control }
        force_update_dynamic_modifier = yes
    }
}
