# HOI4 War Planner Oracle O1 quick-trial helper.
# One command arms a six-hour capture and auto-stops after the hour-6 sample.
# IMPORTANT: same-save reloads replayed the same combat trace on 1.19.3.
# The preburn commands below consume scripted random draws BEFORE combat is created,
# allowing us to test whether battle seeding shares the same RNG stream.

oracle_o1_trial6_start = {
    if = {
        limit = { NOT = { has_variable = oracle_o1_preburn_count } }
        set_variable = { oracle_o1_preburn_count = 0 }
    }
    clr_country_flag = oracle_o1_running
    set_country_flag = oracle_o1_running
    set_variable = { oracle_o1_hour = 0 }
    log = "WPO1 BEGIN schema=1 scenario=o1-base-neutral-tactics-v1 gameVersion=1.19.3.0.c01a checksum=5632 checksumScope=base-game-reference method=bisection14 runMode=trial6 tacticMode=neutral-basic-only preBurn=[?oracle_o1_preburn_count]"
    oracle_o1_capture_sample = yes
    country_event = { id = oracle_o1.2 hours = 1 }
    country_event = { id = oracle_o1.2 hours = 2 }
    country_event = { id = oracle_o1.2 hours = 3 }
    country_event = { id = oracle_o1.2 hours = 4 }
    country_event = { id = oracle_o1.2 hours = 5 }
    country_event = { id = oracle_o1.2 hours = 6 }
}

oracle_o1_rng_burn_once = {
    random_list = {
        1 = { set_temp_variable = { oracle_o1_rng_sink = 0 } }
        1 = { set_temp_variable = { oracle_o1_rng_sink = 1 } }
    }
}

d_oracle_o1_preburn64 = {
    GER = {
        set_variable = { oracle_o1_preburn_count = 64 }
        for_loop_effect = {
            start = 0
            end = 64
            value = oracle_o1_rng_burn_i
            oracle_o1_rng_burn_once = yes
        }
        log = "WPO1 STATUS state=preburn-complete preBurn=64"
    }
}
