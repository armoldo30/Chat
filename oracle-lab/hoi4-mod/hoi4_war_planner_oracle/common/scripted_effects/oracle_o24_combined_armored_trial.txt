            force_update_dynamic_modifier = yes
        }

        force_update_dynamic_modifier = yes
    }
    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o24_attack_target }
                has_tech = oracle_o24_p9
                POL = { has_dynamic_modifier = { modifier = oracle_o24_defender_control } }
            }
            set_country_flag = oracle_o24_prepared
