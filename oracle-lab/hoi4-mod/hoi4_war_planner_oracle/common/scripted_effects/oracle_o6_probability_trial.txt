oracle_o6_remove_attack_modifiers = {
    if = {
        limit = { has_dynamic_modifier = { modifier = oracle_o6_low_attack } }
        remove_dynamic_modifier = { modifier = oracle_o6_low_attack }
        force_update_dynamic_modifier = yes
    }
    if = {
        limit = { has_dynamic_modifier = { modifier = oracle_o6_high_attack } }
        remove_dynamic_modifier = { modifier = oracle_o6_high_attack }
        force_update_dynamic_modifier = yes
    }
}
