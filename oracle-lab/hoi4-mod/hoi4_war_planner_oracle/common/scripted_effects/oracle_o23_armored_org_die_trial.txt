d_oracle_o23_prepare = {
    GER = {
        clr_country_flag = oracle_o23_prepared
        clr_country_flag = oracle_o23_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o23_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o23_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o23_attack_target }
        force_update_dynamic_modifier = yes
        set_technology = { oracle_o23_p20 = 1 popup = no }
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o23_defender_control } }
            remove_dynamic_modifier = { modifier = oracle_o23_defender_control }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o23_defender_control }
        force_update_dynamic_modifier = yes
    }
    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o23_attack_target }
                has_tech = oracle_o23_p20
                POL = { has_dynamic_modifier = { modifier = oracle_o23_defender_control } }
            }
            set_country_flag = oracle_o23_prepared
            log = "WPO23 STATUS state=prepared schema=1 scenario=o23-armored-org-die-v1 expectedPanel=GER-SA20-GER-P20-GER-A20-POL-DEF10-POL-P4-POL-A20 expectedArmorRelation=POL-cannot-pierce-GER expectedOutgoingPiercing=GER-fully-pierces-POL"
        }
        else = { log = "WPO23 STATUS state=prepare-failed" }
    }
}
