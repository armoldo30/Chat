    id = oracle_o4.1
    hidden = yes
    is_triggered_only = yes

    immediate = {

            limit = { has_country_flag = oracle_o4_running }

            add_to_variable = { oracle_o4_hour = 1 }
            oracle_o4_capture_sample = yes

            if = {
                limit = {
                    NOT = {
                        check_variable = {
                            var = oracle_o4_hour
                            value = 6
                            compare = less_than
                        }
                    }
                }

                clr_country_flag = oracle_o4_running

                if = {
                    limit = {
                        has_dynamic_modifier = {
                            modifier = oracle_o4_low_attack_attenuator
                        }
                    }
                    remove_dynamic_modifier = {
                        modifier = oracle_o4_low_attack_attenuator
                    }
                    force_update_dynamic_modifier = yes
                }

                POL = {
                    if = {
                        limit = {
                            has_dynamic_modifier = {
                                modifier = oracle_o4_zero_defense_suppressor
                            }
                        }
                        remove_dynamic_modifier = {
                            modifier = oracle_o4_zero_defense_suppressor
                        }
                        force_update_dynamic_modifier = yes
                    }
                }

                clr_country_flag = oracle_o4_prepared

                if = {
                    limit = {
                        OR = {
                            has_dynamic_modifier = {
                                modifier = oracle_o4_low_attack_attenuator
                            }
                            POL = {
                                has_dynamic_modifier = {
                                    modifier = oracle_o4_zero_defense_suppressor
                                }
                            }
                        }
                    }
                    log = "WPO4 END hour=[?oracle_o4_hour] reason=trial6-complete modifiersRemoved=no cleanupFailure=yes"

                    log = "WPO4 END hour=[?oracle_o4_hour] reason=trial6-complete modifiersRemoved=yes cleanupFailure=no"
                }
            }
        }
    }
