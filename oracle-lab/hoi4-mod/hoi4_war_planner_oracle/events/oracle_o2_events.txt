add_namespace = oracle_o2

# Hidden sampler for d_oracle_o2_trial6.
# Six copies are queued up front at delays 1..6 hours. Do not recursively
# reschedule: that pattern caused same-tick scheduler collapse during early O1 work.
country_event = {
    id = oracle_o2.1
    hidden = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { has_country_flag = oracle_o2_running }

            add_to_variable = { oracle_o2_hour = 1 }
            oracle_o2_capture_sample = yes

            if = {
                limit = {
                    NOT = {
                        check_variable = {
                            var = oracle_o2_hour
                            value = 6
                            compare = less_than
                        }
                    }
                }

                clr_country_flag = oracle_o2_running
                if = {
                            limit = {
                                has_dynamic_modifier = {
                                    modifier = oracle_o2_defended_attack_amplifier
                                }
                            }
                            remove_dynamic_modifier = {
                                modifier = oracle_o2_defended_attack_amplifier
                            }
                        }
                force_update_dynamic_modifier = yes
                clr_country_flag = oracle_o2_prepared
                log = "WPO2 END hour=[?oracle_o2_hour] reason=trial6-complete amplifierRemoved=yes"
            }
        }
    }
}
