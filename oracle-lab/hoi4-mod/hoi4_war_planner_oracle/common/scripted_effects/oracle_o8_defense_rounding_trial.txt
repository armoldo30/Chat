# O8 defense-point integerization probe.
#
# Diagnostic idea:
# - O7 established the wider attack-point distribution at Soft Attack 20.
# - defended points are forced to miss;
# - undefended points are forced to hit;
# - organization dice are fixed to 1 and strength damage to 0.
#
# Therefore each defender organization-loss multiplicity reveals
# max(attackPoints - defensePoints, 0).
#
# Acceptance panel:
# - GER displayed Soft Attack exactly 20, tooltip/effective 20.0;
# - POL displayed Defense exactly 10, tooltip/effective 10.0 if shown;
# - one GER division vs one POL division, width 18 each;
# - no commanders, zero reserves.

oracle_o8_log_attacker_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO8 ATTACKER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o8_log_defender_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO8 DEFENDER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o8_capture_sample = {
    log = "WPO8 SAMPLE hour=[?oracle_o8_hour]"
    oracle_o8_log_attacker_divisions = yes
    POL = {
        oracle_o8_log_defender_divisions = yes
    }
}

d_oracle_o8_prepare = {
    GER = {
        clr_country_flag = oracle_o8_prepared
        clr_country_flag = oracle_o8_running

        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o8_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o8_attack_target }
            force_update_dynamic_modifier = yes
        }

        add_dynamic_modifier = { modifier = oracle_o8_attack_target }
        force_update_dynamic_modifier = yes
    }

    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o8_defense_target } }
            remove_dynamic_modifier = { modifier = oracle_o8_defense_target }
            force_update_dynamic_modifier = yes
        }

        add_dynamic_modifier = { modifier = oracle_o8_defense_target }
        force_update_dynamic_modifier = yes
    }

    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o8_attack_target }
                POL = { has_dynamic_modifier = { modifier = oracle_o8_defense_target } }
            }
            set_country_flag = oracle_o8_prepared
            log = "WPO8 STATUS state=prepared scenario=o8-defense-wide-rounding-v1 attackModifier=army_infantry_attack_factor:-0.721 defenseModifier=army_infantry_defence_factor:-0.9607843137 modifiersPresent=yes expectedPanel=GER-soft-20-tooltip-20.0-POL-defense-10"
        }
        else = {
            log = "WPO8 STATUS state=prepare-failed"
        }
    }
}

d_oracle_o8_clear = {
    GER = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o8_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o8_attack_target }
            force_update_dynamic_modifier = yes
        }
        clr_country_flag = oracle_o8_prepared
        clr_country_flag = oracle_o8_running
    }

    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o8_defense_target } }
            remove_dynamic_modifier = { modifier = oracle_o8_defense_target }
            force_update_dynamic_modifier = yes
        }
    }

    GER = {
        if = {
            limit = {
                OR = {
                    has_dynamic_modifier = { modifier = oracle_o8_attack_target }
                    POL = { has_dynamic_modifier = { modifier = oracle_o8_defense_target } }
                }
            }
            log = "WPO8 STATUS state=clear-failed modifiersRemoved=no"
        }
        else = {
            log = "WPO8 STATUS state=cleared modifiersRemoved=yes"
        }
    }
}
