d_oracle_o7_prepare = {
    GER = {
        clr_country_flag = oracle_o7_prepared
        clr_country_flag = oracle_o7_running

        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o7_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o7_attack_target }
            force_update_dynamic_modifier = yes
        }

        add_dynamic_modifier = { modifier = oracle_o7_attack_target }
        force_update_dynamic_modifier = yes

        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o7_attack_target } }
            set_country_flag = oracle_o7_prepared
            log = "WPO7 STATUS state=prepared scenario=o7-wide-random-rounding-v1 attackModifier=army_infantry_attack_factor:-0.721 modifierPresent=yes expectedPanel=GER-soft-20-tooltip-19.8-to-20.2-POL-defense-high"
        }
        else = { log = "WPO7 STATUS state=prepare-failed" }
    }
}
