d_oracle_o18_prepare = {
    GER = {
        clr_country_flag = oracle_o18_prepared
        clr_country_flag = oracle_o18_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o18_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o18_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o18_attack_target }
        force_update_dynamic_modifier = yes
    }

    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o18_defender_control } }
            remove_dynamic_modifier = { modifier = oracle_o18_defender_control }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o18_defender_control }
        force_update_dynamic_modifier = yes
    }

    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o18_attack_target }
                POL = { has_dynamic_modifier = { modifier = oracle_o18_defender_control } }
            }
            set_country_flag = oracle_o18_prepared
            log = "WPO18 STATUS state=prepared scenario=o18-normal-strength-die-v1 attackModifier=army_infantry_attack_factor:-0.721 defenderAttackModifier=army_infantry_attack_factor:-0.99 defenseModifier=army_infantry_defence_factor:-0.9923784016 modifiersPresent=yes expectedPanel=GER-soft-20-tooltip-20.0-POL-defense-10-tooltip-10.0"
        }
        else = { log = "WPO18 STATUS state=prepare-failed" }
    }
}

d_oracle_o18_clear = {
    GER = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o18_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o18_attack_target }
            force_update_dynamic_modifier = yes
        }
        clr_country_flag = oracle_o18_prepared
        clr_country_flag = oracle_o18_running
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o18_defender_control } }
            remove_dynamic_modifier = { modifier = oracle_o18_defender_control }
            force_update_dynamic_modifier = yes
        }
    }
    GER = {
        if = {
            limit = {
                OR = {
                    has_dynamic_modifier = { modifier = oracle_o18_attack_target }
                    POL = { has_dynamic_modifier = { modifier = oracle_o18_defender_control } }
                }
            }
            log = "WPO18 STATUS state=clear-failed modifiersRemoved=no"
        }
        else = { log = "WPO18 STATUS state=cleared modifiersRemoved=yes" }
    }
}
