add_namespace = oracle_o1

# Hidden event used only by d_oracle_o1_trial6.
# Six copies are queued up front at delays 1..6 hours so we preserve the scheduler behavior
# already validated against HOI4 1.19.2 and avoid recursive same-tick collapse.
country_event = {
    id = oracle_o1.2
    hidden = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { has_country_flag = oracle_o1_running }

            add_to_variable = { oracle_o1_hour = 1 }
            oracle_o1_capture_sample = yes

            if = {
                limit = {
                    NOT = {
                        check_variable = {
                            var = oracle_o1_hour
                            value = 6
                            compare = less_than
                        }
                    }
                }
                clr_country_flag = oracle_o1_running
                log = "WPO1 END hour=[?oracle_o1_hour] reason=trial6-complete"
            }
        }
    }
}
