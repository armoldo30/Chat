d_oracle_o19_prepare = {
    GER = {
        clr_country_flag = oracle_o19_prepared
        clr_country_flag = oracle_o19_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o19_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o19_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o19_attack_target }
        force_update_dynamic_modifier = yes
    }

    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o19_defender_control } }
            remove_dynamic_modifier = { modifier = oracle_o19_defender_control }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o19_defender_control }
        force_update_dynamic_modifier = yes
    }

    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o19_attack_target }
                POL = { has_dynamic_modifier = { modifier = oracle_o19_defender_control } }
            }
            set_country_flag = oracle_o19_prepared
            log = "WPO19 STATUS state=prepared scenario=o19-combined-normal-damage-v1 attackModifier=army_infantry_attack_factor:-0.721 defenderAttackModifier=army_infantry_attack_factor:-0.99 defenseModifier=army_infantry_defence_factor:-0.9923784016 modifiersPresent=yes expectedPanel=GER-soft-20-tooltip-20.0-POL-defense-10-tooltip-10.0"
        }
        else = { log = "WPO19 STATUS state=prepare-failed" }
    }
}

    GER = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o19_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o19_attack_target }
            force_update_dynamic_modifier = yes
        }
        clr_country_flag = oracle_o19_prepared
        clr_country_flag = oracle_o19_running
    }

    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o19_defender_control } }
            remove_dynamic_modifier = { modifier = oracle_o19_defender_control }
            force_update_dynamic_modifier = yes
        }
    }

    GER = {
        if = {
            limit = {
                OR = {
                    has_dynamic_modifier = { modifier = oracle_o19_attack_target }
                    POL = { has_dynamic_modifier = { modifier = oracle_o19_defender_control } }
                }
            }
            log = "WPO19 STATUS state=clear-failed modifiersRemoved=no"
        }
        else = { log = "WPO19 STATUS state=cleared modifiersRemoved=yes" }
    }
}
