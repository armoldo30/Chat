d_oracle_o10v2_prepare = {
    GER = {
        clr_country_flag = oracle_o10v2_prepared
        clr_country_flag = oracle_o10v2_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o10v2_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o10v2_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o10v2_attack_target }
        force_update_dynamic_modifier = yes
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o10v2_minimum_defense } }
            remove_dynamic_modifier = { modifier = oracle_o10v2_minimum_defense }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o10v2_minimum_defense }
        force_update_dynamic_modifier = yes
    }
    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o10v2_attack_target }
                POL = { has_dynamic_modifier = { modifier = oracle_o10v2_minimum_defense } }
            }
            set_country_flag = oracle_o10v2_prepared
            log = "WPO10V2 STATUS state=prepared scenario=o10v2-minimum-defense-response-v1 attackModifier=army_infantry_attack_factor:-0.721 defenseModifier=army_infantry_defence_factor:-1.0328836429 modifiersPresent=yes expectedPanel=GER-soft-20-tooltip-20.0-POL-defense-1.9-tooltip-1.9"
        }
        else = { log = "WPO10V2 STATUS state=prepare-failed" }
    }
}

    }
}
