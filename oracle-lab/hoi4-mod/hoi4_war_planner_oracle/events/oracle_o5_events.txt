add_namespace = oracle_o5

country_event = {
    id = oracle_o5.1
    hidden = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { has_country_flag = oracle_o5_running }

            add_to_variable = { oracle_o5_hour = 1 }
            oracle_o5_capture_sample = yes

            if = {
                limit = {
                    NOT = {
                        check_variable = {
                            var = oracle_o5_hour
                            value = 11
                            compare = less_than
                        }
                    }
                }

                clr_country_flag = oracle_o5_running

                if = {
                    limit = {
                        has_dynamic_modifier = {
                            modifier = oracle_o5_attack_target
                        }
                    }
                    remove_dynamic_modifier = {
                        modifier = oracle_o5_attack_target
                    }
                    force_update_dynamic_modifier = yes
                }

                clr_country_flag = oracle_o5_prepared

                if = {
                    limit = {
                        has_dynamic_modifier = {
                            modifier = oracle_o5_attack_target
                        }
                    }
                    log = "WPO5 END hour=[?oracle_o5_hour] reason=trial11-complete modifierRemoved=no cleanupFailure=yes"
                }
                else = {
                    log = "WPO5 END hour=[?oracle_o5_hour] reason=trial11-complete modifierRemoved=yes cleanupFailure=no"
                }
            }
        }
    }
}
