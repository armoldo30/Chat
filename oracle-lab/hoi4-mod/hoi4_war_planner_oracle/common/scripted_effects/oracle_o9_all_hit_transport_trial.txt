# O9 all-hit transport control.
#
# Purpose:
# Re-run the accepted O8 20 Soft Attack / 10 Defense matchup while forcing both
# defended and undefended attack points to hit. If the O7 attack-point law
# transports to this low-defense scenario, total defender org-loss multiplicity
# should again be 1x/2x/3x at 25%/50%/25%.
#
# Acceptance:
# - GER displayed/tooltip Soft Attack exactly 20.0
# - POL displayed/tooltip Defense exactly 10.0
# - one GER vs one POL 18-width division
# - no commanders, zero reserves

oracle_o9_log_attacker_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes
        log = "WPO9 ATTACKER orgLow=[?oracle_org_low] orgHigh=[?oracle_org_high] strengthLow=[?oracle_strength_low] strengthHigh=[?oracle_strength_high]"
    }
}

oracle_o9_log_defender_divisions = {
    every_country_division = {
        oracle_o1_measure_division = yes

    }

oracle_o9_capture_sample = {
    log = "WPO9 SAMPLE hour=[?oracle_o9_hour]"
    oracle_o9_log_attacker_divisions = yes
    POL = { oracle_o9_log_defender_divisions = yes }
}

d_oracle_o9_prepare = {
    GER = {
        clr_country_flag = oracle_o9_prepared
        clr_country_flag = oracle_o9_running
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o9_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o9_attack_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o9_attack_target }
        force_update_dynamic_modifier = yes
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o9_defense_target } }
            remove_dynamic_modifier = { modifier = oracle_o9_defense_target }
            force_update_dynamic_modifier = yes
        }
        add_dynamic_modifier = { modifier = oracle_o9_defense_target }
        force_update_dynamic_modifier = yes
    }
    GER = {
        if = {
            limit = {
                has_dynamic_modifier = { modifier = oracle_o9_attack_target }
                POL = { has_dynamic_modifier = { modifier = oracle_o9_defense_target } }
            }
            set_country_flag = oracle_o9_prepared
            log = "WPO9 STATUS state=prepared scenario=o9-o8-all-hit-transport-v1 attackModifier=army_infantry_attack_factor:-0.721 defenseModifier=army_infantry_defence_factor:-0.9923784016 modifiersPresent=yes expectedPanel=GER-soft-20-tooltip-20.0-POL-defense-10-tooltip-10.0"
        }
        else = { log = "WPO9 STATUS state=prepare-failed" }
    }
}

d_oracle_o9_clear = {
    GER = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o9_attack_target } }
            remove_dynamic_modifier = { modifier = oracle_o9_attack_target }
            force_update_dynamic_modifier = yes
        }
        clr_country_flag = oracle_o9_prepared
        clr_country_flag = oracle_o9_running
    }
    POL = {
        if = {
            limit = { has_dynamic_modifier = { modifier = oracle_o9_defense_target } }
            remove_dynamic_modifier = { modifier = oracle_o9_defense_target }
            force_update_dynamic_modifier = yes
        }
    }
    GER = {
        if = {
            limit = {
                OR = {
                    has_dynamic_modifier = { modifier = oracle_o9_attack_target }
                    POL = { has_dynamic_modifier = { modifier = oracle_o9_defense_target } }
                }
            }
            log = "WPO9 STATUS state=clear-failed modifiersRemoved=no"
        }
        else = { log = "WPO9 STATUS state=cleared modifiersRemoved=yes" }
    }
}
